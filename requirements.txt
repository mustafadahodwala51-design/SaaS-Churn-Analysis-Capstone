# Runtime dependencies for the SaaS churn capstone.
#
# Install everything with:
#     pip install -r requirements.txt
#
# Versions are pinned to the environment the project was built and verified in
# (Python 3.12.10), so a fresh install reproduces the same outputs. The data
# generator is seeded, so results do not depend on library versions - but
# pinning avoids surprises if a future pandas changes default parsing
# behaviour, which is exactly how the mixed-date bug in this project behaved.

# --- Core analysis ---------------------------------------------------------
pandas==3.0.5
numpy==2.5.3

# --- Charts ----------------------------------------------------------------
matplotlib==3.11.2
seaborn==0.13.2

# --- Excel deliverable -----------------------------------------------------
# build_excel.py uses openpyxl to format the workbook and embed the PNG charts.
openpyxl==3.1.5

# --- SQL layer -------------------------------------------------------------
# analysis.sql is written for DuckDB, which reads the CSVs directly, so there is
# no database server to install. Also required by validate_sql.py.
duckdb==1.5.5

# --- Notebook deliverable --------------------------------------------------
# build_notebook.py assembles the .ipynb. ipykernel is needed only to execute
# it (Kernel > Restart & Run All), not just to read it.
nbformat==5.11.1
nbclient==0.11.0
ipykernel==7.3.0
